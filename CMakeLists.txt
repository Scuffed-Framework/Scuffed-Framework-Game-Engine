cmake_minimum_required(VERSION 4.0)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

    set(CMAKE_PREFIX_PATH ${CMAKE_PREFIX_PATH} CACHE STRING "" FORCE)

if(MSVC)
    # Copy compile_commands to source root so VS/IntelliSense can find it
    add_custom_target(compile_commands ALL
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_BINARY_DIR}/compile_commands.json"
            "${CMAKE_SOURCE_DIR}/compile_commands.json"
        COMMENT "Copying compile_commands.json to source root"
    )
endif()

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(_CONAN_TOOLCHAIN "${CMAKE_SOURCE_DIR}/build/${CMAKE_BUILD_TYPE}/generators/conan_toolchain.cmake")
    if(EXISTS "${_CONAN_TOOLCHAIN}")
        set(CMAKE_TOOLCHAIN_FILE "${_CONAN_TOOLCHAIN}" CACHE STRING "Conan toolchain file")
    else()
        message(FATAL_ERROR "conan_toolchain.cmake not found at: ${_CONAN_TOOLCHAIN}\nRun: conan install . --profile=./profiles/windows-x64-debug.txt --build=missing")
    endif()
endif()

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL") # got the sars-copilot-2 virus while reading this
    cmake_policy(SET CMP0091 NEW)
endif()

project("SF"
    VERSION 1.0.0
    DESCRIPTION "Games? (AI hallucination goes here)"
    LANGUAGES C CXX
)

if(CMAKE_EXPORT_COMPILE_COMMANDS)
    set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_INCLUDES OFF)
    set(CMAKE_C_USE_RESPONSE_FILE_FOR_INCLUDES OFF)
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(ARCH_SUFFIX "x64")
else()
    set(ARCH_SUFFIX "x86")
endif()

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Debug")
endif()

set(BASE_OUTPUT_DIR "${CMAKE_SOURCE_DIR}/000-Build-${CMAKE_BUILD_TYPE}-${ARCH_SUFFIX}")

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${BASE_OUTPUT_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${BASE_OUTPUT_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${BASE_OUTPUT_DIR}")
set(CMAKE_PDB_OUTPUT_DIRECTORY "${BASE_OUTPUT_DIR}")

# Multi-config generators (Visual Studio)
if(CMAKE_CONFIGURATION_TYPES)
    foreach(CONFIG ${CMAKE_CONFIGURATION_TYPES})
        string(TOUPPER ${CONFIG} CONFIG_UPPER)
        set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${BASE_OUTPUT_DIR}")
        set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${BASE_OUTPUT_DIR}")
        set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${BASE_OUTPUT_DIR}")
        set(CMAKE_PDB_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${BASE_OUTPUT_DIR}")
    endforeach()
endif()

if (MSVC)
    add_compile_options(
        /Zc:__cplusplus
        /EHsc
        /utf-8
        /wd4005
        /wd4224
        /wd4267
        /wd4244
        /wd4002
        $<$<CONFIG:Debug>:/Zi>
        $<$<CONFIG:Release>:/O2>
    )

    add_link_options(
        /ignore:4099
        /ignore:4098
        /ignore:4217
        /ignore:4286
    )

elseif (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(
        -Wall
        -Wextra
        -Wpedantic
    )
endif()
# Ensure Conan generators path is available to all subdirectories
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/build/${CMAKE_BUILD_TYPE}/generators")
set(CMAKE_PREFIX_PATH ${CMAKE_PREFIX_PATH} CACHE STRING "" FORCE)

add_subdirectory(
    "${CMAKE_SOURCE_DIR}/SF_Engine"                    # SF_Engine source folder
    "${BASE_OUTPUT_DIR}/SF_Engine_Build"               # SF_Engine build folder (outside source!)
)

add_subdirectory(
    "${CMAKE_SOURCE_DIR}/SF_Editor"                
    "${BASE_OUTPUT_DIR}/SF_Editor_Build"           
)

set(VERSION_HEADER "./SF_Engine/Engine/Version.hpp")

set(BUILD_FILE "./build_number.txt")

if(EXISTS "${BUILD_FILE}")
    file(READ "${BUILD_FILE}" LAST_BUILD)
    math(EXPR BUILD_NUMBER "${LAST_BUILD} + 1")
else()
    set(BUILD_NUMBER 0)
endif()

# TODO: Move to powershell so build number can be increased when built with ./scripts/build.ps1 ; and reset build number every month
file(WRITE "${BUILD_FILE}" "${BUILD_NUMBER}")

string(TIMESTAMP CURRENT_YEAR "%Y")
string(TIMESTAMP CURRENT_MONTH "%m")

file(WRITE "${VERSION_HEADER}" "// Auto-generated by CMake\n")
file(APPEND "${VERSION_HEADER}" "#pragma once\n\n")
file(APPEND "${VERSION_HEADER}" "#include <cstdint>\n#include <string_view>\n\n")
file(APPEND "${VERSION_HEADER}" "namespace SF::Engine {\n")
file(APPEND "${VERSION_HEADER}" "constexpr uint16_t Engine_VERSION_MAJOR =  ${CURRENT_YEAR};\n")
file(APPEND "${VERSION_HEADER}" "constexpr uint8_t Engine_VERSION_MINOR = ${CURRENT_MONTH};\n")
file(APPEND "${VERSION_HEADER}" "constexpr uint16_t Engine_VERSION_PATCH = ${BUILD_NUMBER};\n")
file(APPEND "${VERSION_HEADER}" "constexpr std::string_view Engine_VERSION = \"${CURRENT_YEAR}.${CURRENT_MONTH}.${BUILD_NUMBER}\";\n")
file(APPEND "${VERSION_HEADER}" "} // namespace SF::Engine\n")
